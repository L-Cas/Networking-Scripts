def atomic_write_json(path: Path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    with tmp.open("w", encoding="utf-8") as f:
        json.dump(data, f, indent=2, ensure_ascii=False)
    tmp.replace(path)


def load_json(path: Path, default):
    if not path.exists():
        atomic_write_json(path, default)
        return default
    try:
        with path.open("r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return default

def safe_filename(value: str) -> str:
    value = re.sub(r'[<>:"/\\|?*]+', "_", str(value or "").strip())
    value = re.sub(r"\s+", "_", value)
    return value or "device"

    device = device or {}
    return {

def describe_storage_level(level_def):
    labels = []
    for part in level_def.get("parts", []):
        if part.get("type") == "token":
            labels.append(f"[{STORAGE_TOKEN_LABELS.get(part.get('value'), part.get('value', 'Field'))}]")
        else:
            value = str(part.get("value", ""))
            if value:
                labels.append(repr(value))
    return " + ".join(labels) if labels else "(empty folder)"


def render_storage_level(level_def, values):
    pieces = []
    for part in level_def.get("parts", []):
        if part.get("type") == "token":
            pieces.append(str(values.get(part.get("value"), "Unspecified")))
        elif part.get("type") == "text":
            pieces.append(str(part.get("value", "")))
    # Sanitize after composing so separators such as '_' and '-' remain intentional.
    return safe_filename("".join(pieces) or "Unspecified")


def build_storage_path(base: Path, levels, values):
    path = Path(base)
    for level_def in levels:
        path = path / render_storage_level(level_def, values)
    return path


def build_storage_path_to_depth(base: Path, levels, depth: int, values):
    """Resolve a save point where depth=0 is Base and depth=N is after N folder levels."""
    depth = max(0, min(int(depth), len(levels)))
    return build_storage_path(base, levels[:depth], values)


def storage_level_tokens(level_def):
    return {
        part.get("value")
        for part in level_def.get("parts", [])
        if part.get("type") == "token"
    }


def valid_collection_storage_depths(levels):
    """Collection-wide files cannot resolve paths after a device-scoped folder level."""
    valid = [0]
    for idx, level_def in enumerate(levels, 1):
        if storage_level_tokens(level_def) & DEVICE_SCOPED_STORAGE_TOKENS:
            break
        valid.append(idx)
    return valid


def normalize_storage_layout(layout):
    """Migrate v0.8 run/device layouts into the v0.9 single-hierarchy save-point model."""
    layout = clone_layout(layout or {})
    if "levels" in layout:
        levels = layout.get("levels", [])
        dest = layout.setdefault("destinations", {})
        count = len(levels)
        dest.setdefault("manifest", min(count, 0))
        dest.setdefault("summary", dest.get("manifest", 0))
        dest.setdefault("device_files", count)
        for key in ("manifest", "summary", "device_files"):
            try:
                dest[key] = max(0, min(int(dest.get(key, 0)), count))
            except Exception:
                dest[key] = count if key == "device_files" else 0
        return layout

    run_levels = layout.get("run_levels", [])
    device_levels = layout.get("device_levels", [])
    levels = clone_layout(run_levels) + clone_layout(device_levels)
    run_depth = len(run_levels)
    return {
        "levels": levels,
        "destinations": {
            "manifest": run_depth,
            "summary": run_depth,
            "device_files": len(levels),
        },
    }


def storage_depth_label(depth, levels):
    if int(depth) == 0:
        return "Base Folder"
    idx = int(depth) - 1
    if 0 <= idx < len(levels):
        return f"After {depth} · {describe_storage_level(levels[idx])}"
    return f"After {depth} folder levels"


def common_artifact_directory(paths, fallback):
    dirs = [str(Path(x).parent) for x in paths if x]
    if not dirs:
        return Path(fallback)
    try:
        return Path(os.path.commonpath(dirs))
    except Exception:
        return Path(fallback)


def local_timestamp():
    """Timezone-aware local ISO-8601 timestamp with millisecond precision."""
    return datetime.now().astimezone().isoformat(timespec="milliseconds")


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_file(path: Path) -> str:
    return sha256_bytes(path.read_bytes())



def normalize_cli_command(command: str) -> str:
    """Normalize spacing/case only for command identity checks; never modifies saved output."""
    return " ".join(str(command or "").strip().lower().split())


def is_full_running_config_command(command: str) -> bool:
    return normalize_cli_command(command) in RUNNING_CONFIG_COMMANDS


def build_running_config_cfg(output: str) -> str:
    """
    Build the standalone recovery .cfg from the already-preserved command output.

    commands[].output remains untouched. For the convenience .cfg only, remove Cisco's
    display-only preamble lines that are not configuration statements:
        Building configuration...
        Current configuration : <n> bytes

    The remaining configuration body is preserved line-for-line and terminated with a newline.
    """
    lines = str(output or "").splitlines()
    while lines and not lines[0].strip():
        lines.pop(0)

    while lines:
        stripped = lines[0].strip()
        low = stripped.lower()
        if low == "building configuration..." or re.match(r"^current configuration\s*:\s*\d+\s*bytes\s*$", low):
            lines.pop(0)
            while lines and not lines[0].strip():
                lines.pop(0)
            continue
        break

    cfg = "\n".join(lines)
    if cfg and not cfg.endswith("\n"):
        cfg += "\n"
    return cfg


def find_running_config_entry(result: dict):
    """Return the first successful full running-config command entry, if one exists."""
    for entry in result.get("commands", []):
        if entry.get("status") == "success" and is_full_running_config_command(entry.get("command", "")):
            return entry
    return None


def command_set_sha256(commands) -> str:
    canonical = json.dumps(list(commands), ensure_ascii=False, separators=(",", ":"))
    return sha256_bytes(canonical.encode("utf-8"))


def classify_command_exception(exc) -> str:
    text = f"{exc.__class__.__name__} {exc}".lower()
    return "timeout" if "timeout" in text or "timed out" in text else "execution_error"


def normalize_header(value: str) -> str:
    return re.sub(r"[^a-z0-9]+", "", str(value or "").strip().lower())


def profile_files():
    return sorted(PROFILE_DIR.glob("*.txt"), key=lambda p: p.name.lower())


def load_profile(profile_path: Path):
    commands = []
    for raw in profile_path.read_text(encoding="utf-8-sig").splitlines():
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        commands.append(line)

    metadata = {}
    meta_path = profile_path.with_suffix(".json")
    if meta_path.exists():
        metadata = load_json(meta_path, {})
    return commands, metadata


def is_read_only(command: str) -> bool:
    low = command.strip().lower()
    return any(low == p.strip() or low.startswith(p) for p in READ_ONLY_PREFIXES)


def cli_output_indicates_error(output: str) -> bool:
    low = output.lower()
    return any(pattern.lower() in low for pattern in COMMAND_ERROR_PATTERNS)


def ping_host(host: str, timeout_ms: int = 2000):
    system = platform.system().lower()
    if system == "windows":
        cmd = ["ping", "-n", "1", "-w", str(timeout_ms), host]
    else:
        seconds = max(1, int(round(timeout_ms / 1000)))
        cmd = ["ping", "-c", "1", "-W", str(seconds), host]

    started = time.perf_counter()
    try:
        proc = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            timeout=max(3, timeout_ms / 1000 + 2),
            creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0),
        )
        duration = time.perf_counter() - started
        text = (proc.stdout or "") + (proc.stderr or "")
        return proc.returncode == 0, duration, text.strip()
    except Exception as exc:
        return False, time.perf_counter() - started, str(exc)


def tcp_22_test(host: str, timeout: float = 5.0):
    started = time.perf_counter()
    try:
        with socket.create_connection((host, 22), timeout=timeout):
            pass
        return True, time.perf_counter() - started, ""
    except Exception as exc:
        return False, time.perf_counter() - started, str(exc)

def read_device_workbook(path: Path):
    if load_workbook is None:
        raise RuntimeError("openpyxl is not installed")

    wb = load_workbook(path, data_only=True)
    if "Devices" in wb.sheetnames:
        ws = wb["Devices"]
    else:
        ws = wb[wb.sheetnames[0]]

    header_map = {}
    for col in range(1, ws.max_column + 1):
        header_map[normalize_header(ws.cell(row=1, column=col).value)] = col

    aliases = {
        "name": ["namealias", "name", "alias", "hostname", "device"],
        "ip": ["ipaddress", "ip", "address", "host"],
        "location": ["location", "site", "group"],
        "role": ["role", "type", "devicefunction"],
        "device_type": ["devicetype", "netmikodevicetype", "netmikotype"],
        "notes": ["notes", "description", "comments"],
    }

    resolved = {}
    for key, candidates in aliases.items():
        for candidate in candidates:
            if candidate in header_map:
                resolved[key] = header_map[candidate]
                break

    if "name" not in resolved or "ip" not in resolved:
        raise ValueError("Excel file must contain Name / Alias and IP Address columns.")

    devices = []
    for row in range(2, ws.max_row + 1):
        name = str(ws.cell(row=row, column=resolved["name"]).value or "").strip()
        ip = str(ws.cell(row=row, column=resolved["ip"]).value or "").strip()
        if not name and not ip:
            continue
        if not name or not ip:
            continue

        def cell_value(key, default=""):
            col = resolved.get(key)
            if not col:
                return default
            return str(ws.cell(row=row, column=col).value or default).strip()

        devices.append({
            "name": name,
            "ip": ip,
            "location": cell_value("location"),
            "role": cell_value("role"),
            "device_type": cell_value("device_type", "cisco_ios") or "cisco_ios",
            "notes": cell_value("notes"),
        })

    return devices

        ).grid(row=5, column=0, columnspan=2, sticky="w", pady=(12, 10))

        )

    def refresh_list(self):
        self.listbox.delete(0, "end")
        default = self.data.get("default_profile", self.data.get("general_profile", ""))
        for name in sorted(self.data.get("presets", {}).keys(), key=str.lower):
            marker = "  [Default]" if name == default else ""
            self.listbox.insert("end", name + marker)

    def selected_name(self):
        sel = self.listbox.curselection()
        if not sel:
            return None
        return self.listbox.get(sel[0]).replace("  [Default]", "")

    def load_selected(self, _event=None):
        name = self.selected_name()
        if not name:
            return
        preset = self.data.get("presets", {}).get(name, {})
        self.name_var.set(name)
        self.user_var.set(preset.get("username", ""))
        self.pass_var.set(preset.get("password_base", ""))
        self.desc_var.set(preset.get("description", ""))
        default = self.data.get("default_profile", self.data.get("general_profile", ""))
        self.default_var.set(name == default)

    def new_preset(self):
        self.name_var.set("")
        self.user_var.set("")
        self.pass_var.set("")
        self.desc_var.set("")
        self.default_var.set(False)

    def delete_preset(self):
        name = self.selected_name()
        if not name:
            return
        if not messagebox.askyesno("Delete preset", f"Delete credential preset '{name}'?", parent=self):
            return
        self.data.get("presets", {}).pop(name, None)
        default = self.data.get("default_profile", self.data.get("general_profile", ""))
        if default == name:
            self.data["default_profile"] = next(iter(self.data.get("presets", {})), "")
        self.data.pop("general_profile", None)
        atomic_write_json(CREDENTIAL_FILE, self.data)
        self.master_app.credentials = self.data
        self.master_app.refresh_auth_presets()
        self.refresh_list()
        self.new_preset()

    def save_preset(self):
        name = self.name_var.get().strip()
        if not name:
            messagebox.showwarning("Preset name", "Enter a preset name.", parent=self)
            return
        self.data.setdefault("presets", {})[name] = {
            "username": self.user_var.get().strip(),
            "password_base": self.pass_var.get(),
            "description": self.desc_var.get().strip(),
        }
        if self.default_var.get() or not self.data.get("default_profile"):
            self.data["default_profile"] = name
        self.data.pop("general_profile", None)
        atomic_write_json(CREDENTIAL_FILE, self.data)
        self.master_app.credentials = self.data
        self.master_app.refresh_auth_presets()
        self.refresh_list()
        messagebox.showinfo("Saved", f"Credential preset '{name}' saved locally.", parent=self)

class ReorderLevelListbox(tk.Listbox):
    """Listbox that keeps folder-level definitions and supports drag-to-reorder."""
    def __init__(self, master, on_change=None, **kwargs):
        super().__init__(master, **kwargs)
        self.levels = []
        self.on_change = on_change
        self._drag_index = None
        self.bind("<ButtonPress-1>", self._drag_start, add="+")
        self.bind("<B1-Motion>", self._drag_motion, add="+")

    def set_levels(self, levels):
        self.levels = clone_layout(levels)
        self.refresh()

    def get_levels(self):
        return clone_layout(self.levels)

    def refresh(self):
        selected = self.curselection()
        selected_index = selected[0] if selected else None
        self.delete(0, "end")
        for i, level_def in enumerate(self.levels, 1):
            self.insert("end", f"☰  {i}.  {describe_storage_level(level_def)}")
        if selected_index is not None and self.levels:
            selected_index = min(selected_index, len(self.levels) - 1)
            self.selection_set(selected_index)
            self.activate(selected_index)

    def selected_index(self):
        sel = self.curselection()
        return sel[0] if sel else None

    def _drag_start(self, event):
        if not self.levels:
            self._drag_index = None
            return
        idx = self.nearest(event.y)
        if 0 <= idx < len(self.levels):
            self._drag_index = idx

    def _drag_motion(self, event):
        if self._drag_index is None or not self.levels:
            return
        new_index = self.nearest(event.y)
        new_index = max(0, min(new_index, len(self.levels) - 1))
        if new_index == self._drag_index:
            return
        item = self.levels.pop(self._drag_index)
        self.levels.insert(new_index, item)
        self._drag_index = new_index
        self.refresh()
        self.selection_clear(0, "end")
        self.selection_set(new_index)
        self.activate(new_index)
        if self.on_change:
            self.on_change()

        ttk.Button(controls, text="Move Up", command=lambda: self.move_part(-1)).pack(fill="x", pady=4)
        ttk.Button(controls, text="Move Down", command=lambda: self.move_part(1)).pack(fill="x", pady=4)

    def add_token(self, token):
        self.parts.append(token_part(token))
        self.refresh_parts(select=len(self.parts) - 1)

    def add_text(self):
        value = simpledialog.askstring(
            "Add Text",
            "Literal text to include in the folder name (for example _ or -Archive):",
            parent=self,
        )
        if value is not None and value != "":
            self.parts.append(text_part(value))
            self.refresh_parts(select=len(self.parts) - 1)

    def edit_text(self):
        idx = self.selected_part()
        if idx is None or self.parts[idx].get("type") != "text":
            messagebox.showinfo("Edit Text", "Select a text part first.", parent=self)
            return
        old = self.parts[idx].get("value", "")
        value = simpledialog.askstring("Edit Text", "Text:", initialvalue=old, parent=self)
        if value is not None:
            self.parts[idx] = text_part(value)
            self.refresh_parts(select=idx)

    def selected_part(self):
        sel = self.parts_list.curselection()
        return sel[0] if sel else None

    def move_part(self, direction):
        idx = self.selected_part()
        if idx is None:
            return
        new = idx + direction
        if 0 <= new < len(self.parts):
            self.parts[idx], self.parts[new] = self.parts[new], self.parts[idx]
            self.refresh_parts(select=new)

    def remove_part(self):
        idx = self.selected_part()
        if idx is None:
            return
        self.parts.pop(idx)
        self.refresh_parts(select=min(idx, len(self.parts) - 1) if self.parts else None)

    def refresh_parts(self, select=None):
        self.parts_list.delete(0, "end")
        for part in self.parts:
            if part.get("type") == "token":
                label = STORAGE_TOKEN_LABELS.get(part.get("value"), part.get("value", "Field"))
                self.parts_list.insert("end", f"FIELD   [{label}]")
            else:
                self.parts_list.insert("end", f"TEXT     {part.get('value', '')!r}")
        if select is not None and 0 <= select < len(self.parts):
            self.parts_list.selection_set(select)
            self.parts_list.activate(select)

        example_now = datetime(2026, 10, 19, 19, 22).astimezone()
        values = storage_token_values(
            example_now,
            "General Switch",
            "20261019_192200",
            {"name": "ACC01", "ip": "10.20.30.40", "location": "Building A", "role": "Access"},
        )
        self.preview_var.set(render_storage_level({"parts": self.parts}, values) if self.parts else "(empty)")

    def save(self):
        if not self.parts:
            messagebox.showwarning("Folder Level", "Add at least one field or text part.", parent=self)
            return
        self.result = {"parts": clone_layout(self.parts)}
        self.destroy()

    def _placement_row(self, parent, label, variable):
        frame = ttk.Frame(parent)

    def refresh_preset_values(self):
        values = list(STORAGE_LAYOUT_PRESETS.keys()) + sorted(self.custom_presets.keys(), key=str.lower) + ["Custom"]
        self.preset_combo.configure(values=values)
        if self.preset_var.get() not in values:
            self.preset_var.set("Custom")

    def _depth_map(self, depths):
        levels = self.level_list.get_levels()
        return {storage_depth_label(depth, levels): depth for depth in depths}

    @staticmethod
    def _choose_label(mapping, desired_depth):
        if not mapping:
            return ""
        for label, depth in mapping.items():
            if depth == desired_depth:
                return label
        # If a reorder makes a collection-wide destination invalid, fall back to the
        # deepest valid save point at or above the previous depth.
        allowed = sorted(set(mapping.values()))
        candidates = [x for x in allowed if x <= desired_depth]
        target = max(candidates) if candidates else min(allowed)
        for label, depth in mapping.items():
            if depth == target:
                return label
        return next(iter(mapping))

    def selected_destinations(self):
        return {
            "manifest": self._manifest_map.get(self.manifest_dest_var.get(), 0),
            "summary": self._summary_map.get(self.summary_dest_var.get(), 0),
            "device_files": self._device_map.get(self.device_dest_var.get(), len(self.level_list.get_levels())),
        }

    def refresh_destination_choices(self, desired=None):
        desired = desired or self.selected_destinations()
        levels = self.level_list.get_levels()
        global_depths = valid_collection_storage_depths(levels)
        all_depths = list(range(0, len(levels) + 1))

        self._manifest_map = self._depth_map(global_depths)
        self._summary_map = self._depth_map(global_depths)
        self._device_map = self._depth_map(all_depths)

        self.manifest_combo.configure(values=list(self._manifest_map.keys()))
        self.summary_combo.configure(values=list(self._summary_map.keys()))
        self.device_combo.configure(values=list(self._device_map.keys()))

        self.manifest_dest_var.set(self._choose_label(self._manifest_map, int(desired.get("manifest", 0))))
        self.summary_dest_var.set(self._choose_label(self._summary_map, int(desired.get("summary", 0))))
        self.device_dest_var.set(self._choose_label(self._device_map, int(desired.get("device_files", len(levels)))))

    def current_layout(self):
        return {
            "levels": self.level_list.get_levels(),
            "destinations": self.selected_destinations(),
        }

    def on_layout_changed(self):
        old_dest = self.selected_destinations()
        self.layout = self.current_layout()
        self.preset_var.set("Custom")
        self.refresh_destination_choices(old_dest)
        self.update_preview()

    def on_destination_changed(self):
        self.layout = self.current_layout()
        self.preset_var.set("Custom")
        self.update_preview()

    def on_preset_selected(self, _event=None):
        name = self.preset_var.get()
        if name == "Custom":
            return
        source = STORAGE_LAYOUT_PRESETS.get(name) or self.custom_presets.get(name)
        if source:
            self.layout = normalize_storage_layout(source)
            self.level_list.set_levels(self.layout.get("levels", []))
            self.refresh_destination_choices(self.layout.get("destinations", {}))
            self.update_preview()

    def browse_base(self):
        selected = filedialog.askdirectory(
            title="Choose Collection Base Folder",
            initialdir=self.base_var.get() or str(OUTPUT_DIR),
        )
        if selected:
            self.base_var.set(selected)

    def add_level(self):
        dialog = FolderLevelDialog(self, STORAGE_TOKENS, title="Add Folder Level")
        self.wait_window(dialog)
        if dialog.result:
            old_dest = self.selected_destinations()
            self.level_list.levels.append(dialog.result)
            self.level_list.refresh()
            self.level_list.selection_set(len(self.level_list.levels) - 1)
            self.preset_var.set("Custom")
            self.refresh_destination_choices(old_dest)
            self.update_preview()

    def edit_level(self):
        idx = self.level_list.selected_index()
        if idx is None:
            return
        dialog = FolderLevelDialog(
            self, STORAGE_TOKENS, initial=self.level_list.levels[idx], title="Edit Folder Level"
        )
        self.wait_window(dialog)
        if dialog.result:
            old_dest = self.selected_destinations()
            self.level_list.levels[idx] = dialog.result
            self.level_list.refresh()
            self.level_list.selection_set(idx)
            self.preset_var.set("Custom")
            self.refresh_destination_choices(old_dest)
            self.update_preview()

    def remove_level(self):
        idx = self.level_list.selected_index()
        if idx is None:
            return
        old_dest = self.selected_destinations()
        self.level_list.levels.pop(idx)
        self.level_list.refresh()
        self.preset_var.set("Custom")
        # Destinations are depth-based. Clamp them after removing a folder level.
        max_depth = len(self.level_list.levels)
        old_dest = {k: min(v, max_depth) for k, v in old_dest.items()}
        self.refresh_destination_choices(old_dest)
        self.update_preview()

    def move_level(self, direction):
        idx = self.level_list.selected_index()
        if idx is None:
            return
        new = idx + direction
        if 0 <= new < len(self.level_list.levels):
            old_dest = self.selected_destinations()
            self.level_list.levels[idx], self.level_list.levels[new] = self.level_list.levels[new], self.level_list.levels[idx]
            self.level_list.refresh()
            self.level_list.selection_clear(0, "end")
            self.level_list.selection_set(new)
            self.level_list.activate(new)
            self.preset_var.set("Custom")
            self.refresh_destination_choices(old_dest)
            self.update_preview()

    def save_custom_preset(self):
        name = simpledialog.askstring("Save Storage Preset", "Preset name:", parent=self)
        if not name:
            return
        if name in STORAGE_LAYOUT_PRESETS:
            messagebox.showwarning("Preset Name", "That name is reserved for a built-in preset.", parent=self)
            return
        self.custom_presets[name] = self.current_layout()
        self.refresh_preset_values()
        self.preset_var.set(name)

    def reset_default(self):
        self.layout = normalize_storage_layout(DEFAULT_STORAGE_LAYOUT)
        self.level_list.set_levels(self.layout["levels"])
        self.refresh_destination_choices(self.layout["destinations"])
        self.preset_var.set("By Collection")
        self.update_preview()

    def update_preview(self):
        now = datetime.now().astimezone()
        collection_id = now.strftime("%Y%m%d_%H%M%S")
        device = {"name": "ACC01", "ip": "10.20.30.40", "location": "Building A", "role": "Access"}
        values = storage_token_values(now, "General Switch", collection_id, device)
        base = Path(self.base_var.get() or str(OUTPUT_DIR))
        levels = self.level_list.get_levels()
        dest = self.selected_destinations()
        max_depth = max(dest.values()) if dest else 0

        lines = [str(base)]
        indent = ""
        for depth in range(0, max_depth + 1):
            file_indent = indent + "    " if depth else "    "
            if dest.get("manifest") == depth:
                lines.append(f"{file_indent}[Manifest] Collection_Manifest_General_Switch_{collection_id}.json")
            if dest.get("summary") == depth:
                lines.append(f"{file_indent}[Summary ] Collection_Summary_General_Switch_{collection_id}.log")
            if dest.get("device_files") == depth:
                lines.append(f"{file_indent}[Device  ] ACC01_General_Switch_{collection_id}.json")
                lines.append(f"{file_indent}[Device  ] ACC01_General_Switch_{collection_id}.log")
                lines.append(f"{file_indent}[Config  ] ACC01_running-config_{collection_id}.cfg")
            if depth < max_depth and depth < len(levels):
                name = render_storage_level(levels[depth], values)
                indent += "    "
                lines.append(f"{indent}└── {name}")

        unused = len(levels) - max_depth
        if unused > 0:
            lines.extend(["", f"Note: {unused} folder level(s) are below every save point and will not be created."])

        self.preview_text.configure(state="normal")
        self.preview_text.delete("1.0", "end")
        self.preview_text.insert("1.0", "\n".join(lines))
        self.preview_text.configure(state="disabled")

    def apply(self):
        base_text = self.base_var.get().strip()
        if not base_text:
            messagebox.showwarning("Base Folder", "Choose a base collection folder.", parent=self)
            return
        try:
            base = Path(base_text).expanduser()
            base.mkdir(parents=True, exist_ok=True)
        except Exception as exc:
            messagebox.showerror(
                "Base Folder",
                f"The base folder could not be created or accessed:\n\n{exc}",
                parent=self,
            )
            return

        layout = normalize_storage_layout(self.current_layout())
        levels = layout["levels"]
        valid_global = set(valid_collection_storage_depths(levels))
        if layout["destinations"]["manifest"] not in valid_global or layout["destinations"]["summary"] not in valid_global:
            messagebox.showerror(
                "Invalid Collection File Placement",
                "Manifest and Summary cannot be placed beneath a device-specific folder level.",
                parent=self,
            )
            return

        max_used = max(layout["destinations"].values()) if layout["destinations"] else 0
        if max_used < len(levels):
            if not messagebox.askyesno(
                "Unused Folder Levels",
                f"{len(levels) - max_used} folder level(s) are below every save point and will not be created.\n\n"
                "Save this layout anyway?",
                parent=self,
            ):
                return

        self.master_app.settings["output_directory"] = str(base)
        self.master_app.settings["storage_layout"] = layout
        self.master_app.settings["storage_layout_name"] = self.preset_var.get() or "Custom"
        self.master_app.settings["storage_custom_presets"] = self.custom_presets
        atomic_write_json(SETTINGS_FILE, self.master_app.settings)
        self.master_app.log_ui(
            f"[SETTINGS] Collection storage layout updated: {self.master_app.settings['storage_layout_name']}"
        )
        self.destroy()

    def refresh_device_tree(self):
        locations = sorted({
            (d.get("location") or d.get("group") or "Unspecified").strip() or "Unspecified"
            for d in self.devices
        })
        values = ["All Locations"] + locations
        current = self.location_var.get() if hasattr(self, "location_var") else "All Locations"
        self.location_combo.configure(values=values)
        if current not in values:
            self.location_var.set("All Locations")

        for iid in self.device_tree.get_children():
            self.device_tree.delete(iid)

        search = self.device_search_var.get().strip().lower()
        loc_filter = self.location_var.get()
        for idx, dev in enumerate(self.devices):
            location = (dev.get("location") or dev.get("group") or "Unspecified").strip() or "Unspecified"
            hay = " ".join([
                str(dev.get("name", "")),
                str(dev.get("ip", "")),
                location,
                str(dev.get("role", "")),
                str(dev.get("device_type", "")),
                str(dev.get("notes", "")),
            ]).lower()
            if search and search not in hay:
                continue
            if loc_filter != "All Locations" and location != loc_filter:
                continue
            self.device_tree.insert(
                "", "end", iid=str(idx),
                values=(
                    dev.get("name", ""),
                    dev.get("ip", ""),
                    location,
                    dev.get("role", ""),
                    dev.get("device_type", "cisco_ios"),
                ),
            )
        if hasattr(self, "selected_count_var"):
            self.update_action_states()

    def refresh_profile_list(self):
        ensure_defaults()
        names = [p.stem for p in profile_files()]
        self.profile_combo.configure(values=names)
        preferred = self.profile_var.get() or self.settings.get("last_command_profile", "")
        if preferred not in names:
            preferred = self.settings.get("last_command_profile", "")
        if preferred not in names:
            preferred = DEFAULT_PROFILE_NAME if DEFAULT_PROFILE_NAME in names else (names[0] if names else "")
        self.profile_var.set(preferred)
        self.update_profile_hint()

    def update_profile_hint(self):
        name = self.profile_var.get()
        path = PROFILE_DIR / f"{name}.txt"
        if not path.exists():
            self.profile_hint_var.set("No command profile selected.")
            return
        commands, meta = load_profile(path)
        desc = meta.get("description", "")
        has_run_cfg = any(is_full_running_config_command(cmd) for cmd in commands)
        if has_run_cfg:
            cfg_note = "CFG backup: profile includes running-config"
        elif self.always_collect_running_config_var.get():
            cfg_note = "CFG backup: running-config auto-added"
        else:
            cfg_note = "CFG backup: only if profile includes running-config"
        hint = f"{len(commands)} command(s)  •  {cfg_note}"
        if desc:
            hint += f"  •  {desc}"
        self.profile_hint_var.set(hint)

    def refresh_auth_presets(self):

        values = names + ["Manual Input"]

            default = self.credentials.get("default_profile", "")
            preferred = default if default in names else (names[0] if names else "Manual Input")
        self.auth_var.set(preferred)

    def import_devices_json(self):
        path = filedialog.askopenfilename(
            title="Import Device Library - JSON",
            filetypes=[("JSON files", "*.json"), ("All files", "*.*")],
        )
        if not path:
            return
        try:
            incoming = json.loads(Path(path).read_text(encoding="utf-8-sig"))
            if isinstance(incoming, dict) and "devices" in incoming:

            self.refresh_device_tree()

        except Exception as exc:
            messagebox.showerror("Import failed", str(exc))

    @staticmethod
    def _sanitize_devices(devices):
        clean = []
        for dev in devices:
            name = str(dev.get("name", dev.get("alias", "")) or "").strip()
            ip = str(dev.get("ip", dev.get("host", "")) or "").strip()
            if not name or not ip:
                continue
            clean.append({
                "name": name,
                "ip": ip,
                "location": str(dev.get("location", dev.get("group", "")) or "").strip(),
                "role": str(dev.get("role", "") or "").strip(),
                "device_type": str(dev.get("device_type", "cisco_ios") or "cisco_ios").strip(),
                "notes": str(dev.get("notes", "") or "").strip(),
            })
        return clean

    def current_commands(self):
        path = PROFILE_DIR / f"{self.profile_var.get()}.txt"
        if not path.exists():
            raise FileNotFoundError(f"Command profile not found: {path}")
        profile_commands, metadata = load_profile(path)
        one_off = list(self.one_off_commands)
        all_commands = profile_commands + one_off

        # Config-backup policy is collector-level rather than profile-level. A profile may
        # explicitly request any recognized full running-config command. If it does not,
        # the persistent setting can append one canonical command for this run only.
        automatically_added_commands = []
        has_full_running_config = any(is_full_running_config_command(cmd) for cmd in all_commands)
        if self.settings.get("always_collect_running_config", True) and not has_full_running_config:
            automatic_command = "show running-config"
            all_commands.append(automatic_command)
            automatically_added_commands.append(automatic_command)

        if self.settings.get("read_only_protection", True):
            blocked = [cmd for cmd in all_commands if not is_read_only(cmd)]
            if blocked:
                raise ValueError(
                    "Read-only protection blocked:\n\n" + "\n".join(blocked) +
                    "\n\nAlpha permits SHOW, DIR, PING, and TRACEROUTE commands only."
                )
        profile_info = {
            "name": self.profile_var.get(),
            "source_file": path.name,
            "profile_file_sha256": sha256_file(path),
            "command_set_sha256": command_set_sha256(all_commands),
            "profile_command_count": len(profile_commands),
            "one_off_command_count": len(one_off),
            "automatically_added_command_count": len(automatically_added_commands),
            "automatically_added_commands": automatically_added_commands,
            "running_config_backup_policy": (
                "automatic" if self.settings.get("always_collect_running_config", True) else "profile_only"
            ),
            "total_command_count": len(all_commands),
            "metadata": metadata,
        }
        return all_commands, metadata, profile_info

    # -------------------- Run-time authentication --------------------

    def prepare_run_credentials(self):
        selected = self.auth_var.get()
        use_token = self.use_token_var.get()
        presets = self.credentials.get("presets", {})

        if selected == "Manual Input":
            dialog = CredentialPrompt(self, manual=True, use_token=use_token)
            if not dialog.result:
                return False
            self._run_credentials = dialog.result
            self._run_credential_source = "Manual Input"
            return True

        preset = presets.get(selected)
        if not preset:
            messagebox.showwarning("Authentication", "Select a saved credential preset or Manual Input.")
            return False
        if not preset.get("username") or not preset.get("password_base"):
            messagebox.showwarning(
                "Authentication",
                f"Credential preset '{selected}' is incomplete.\n\nUse Credentials > Manage Saved Presets to configure it.",
            )
            return False

        if use_token:
            dialog = CredentialPrompt(self, preset_name=selected, preset=preset, manual=False, use_token=True)
            if not dialog.result:
                return False
            self._run_credentials = dialog.result
        else:
            self._run_credentials = {
                "username": preset.get("username", "").strip(),
                "password": preset.get("password_base", ""),
                "token_used": False,
            }
        self._run_credential_source = f"Saved Preset: {selected}"
        return True

    def clear_run_credentials(self):
        self._run_credentials = None
        self._run_credential_source = ""

    # -------------------- Actions --------------------

    def ping_selected(self):
        devices = self.selected_devices()
        if not devices:
            return
        self.start_worker(
            self._ping_worker,
            devices,
            operation_label=f"Pinging {len(devices)} device{'s' if len(devices) != 1 else ''}...",
        )

    def _ping_worker(self, devices):
        self.qlog(f"\n{'='*70}\nPING TEST\n{'='*70}")
        passed = 0
        for i, dev in enumerate(devices, 1):
            ok, elapsed, output = ping_host(dev.get("ip", ""))
            if ok:
                passed += 1
            self.qlog(f"[{'PASS' if ok else 'FAIL'}] {dev.get('name')} ({dev.get('ip')}) - {elapsed:.2f}s")
            if not ok and output:
                tail = output.splitlines()[-1] if output.splitlines() else output
                self.qlog(f"       {tail}")
            self.qprogress(i / len(devices) * 100)
        state = "success" if passed == len(devices) else ("error" if passed == 0 else "warning")
        return {"state": state, "status": f"Ping Complete — {passed}/{len(devices)} responded"}

    def test_ssh_selected(self):
        devices = self.selected_devices()
        if not devices:
            return
        if not self.prepare_run_credentials():
            return
        self.start_worker(
            self._test_ssh_worker,
            devices,
            operation_label=f"Testing SSH on {len(devices)} device{'s' if len(devices) != 1 else ''}...",
        )

    def _test_ssh_worker(self, devices):
        self.qlog(f"\n{'='*70}\nSSH CONNECTION TEST\n{'='*70}")
        passed = 0
        for i, dev in enumerate(devices, 1):
            conn = None
            try:
                if self.settings.get("tcp_22_precheck", True):
                    open22, _elapsed, reason = tcp_22_test(dev["ip"])
                    if not open22:
                        self.qlog(f"[FAIL] {dev.get('name')} ({dev['ip']}) TCP/22 unreachable: {reason}")
                        continue
                conn, info = self.connect_device(dev)
                passed += 1
                self.qlog(f"[PASS] {dev.get('name')} ({dev['ip']}) prompt={info['prompt']} auth={info['credential_source']}")
            except Exception as exc:
                self.qlog(f"[FAIL] {dev.get('name')} ({dev.get('ip')}) - {self.clean_exception(exc)}")
            finally:
                if conn:
                    try:
                        conn.disconnect()
                    except Exception:
                        pass
                self.qprogress(i / len(devices) * 100)
        state = "success" if passed == len(devices) else ("error" if passed == 0 else "warning")
        return {"state": state, "status": f"SSH Test Complete — {passed}/{len(devices)} connected"}

    def run_collection(self):
        devices = self.selected_devices()
        if not devices:
            return
        try:
            commands, metadata, profile_info = self.current_commands()
        except Exception as exc:
            messagebox.showerror("Command profile", str(exc))
            return
        if not commands:
            messagebox.showwarning("No commands", "The selected command profile contains no commands.")
            return
        if not self.prepare_run_credentials():
            return

        now = datetime.now().astimezone()
        batch_id = now.strftime("%Y%m%d_%H%M%S")
        profile_name = self.profile_var.get()
        configured = Path(self.settings.get("output_directory", str(OUTPUT_DIR))).expanduser()
        storage_layout = normalize_storage_layout(self.settings.get("storage_layout", DEFAULT_STORAGE_LAYOUT))
        context = {
            "collection_id": batch_id,
            "profile": profile_name,
            "safe_profile": safe_filename(profile_name),
            "profile_info": profile_info,
            "started": now.isoformat(timespec="milliseconds"),
            "run_datetime": now,
            "storage_base": configured,
            "storage_layout": storage_layout,
            "storage_layout_name": self.settings.get("storage_layout_name", "Custom"),
        }
        self.start_worker(
            self._collection_worker,
            devices,
            commands,
            metadata,
            context,
            operation_label=f"Collecting {len(devices)} device{'s' if len(devices) != 1 else ''}...",
        )

    def connect_device(self, device):
        if ConnectHandler is None:
            raise RuntimeError("Netmiko is not installed. Run: pip install netmiko")
        if not self._run_credentials:
            raise RuntimeError("Run credentials were not prepared.")

        params = {
            "device_type": device.get("device_type", self.settings.get("default_device_type", "cisco_ios")) or "cisco_ios",
            "host": device["ip"],
            "username": self._run_credentials["username"],
            "password": self._run_credentials["password"],
            "fast_cli": False,
        }
        conn = ConnectHandler(**params)
        try:
            prompt = conn.find_prompt().strip()
        except Exception:
            prompt = ""
        return conn, {
            "prompt": prompt,
            "credential_source": self._run_credential_source,
            "username": self._run_credentials["username"],
            "device_type": params["device_type"],
            "token_used": bool(self._run_credentials.get("token_used")),
        }

    def _collection_worker(self, devices, commands, profile_metadata, context):
        self.qlog(f"\n{'='*78}\nCOLLECTION\n{'='*78}")
        self.qlog(f"Collection ID: {context['collection_id']} | Profile: {context['profile']} | Devices: {len(devices)} | Commands/device: {len(commands)}")
        context["storage_base"].mkdir(parents=True, exist_ok=True)
        results = []
        for i, dev in enumerate(devices, 1):
            results.append(self.collect_device(dev, commands, profile_metadata, context))
            self.qprogress(i / len(devices) * 100)

        connected = sum(1 for r in results if r.get("connection", {}).get("success"))
        successful = sum(1 for r in results if self.device_result_status(r) == "SUCCESS")
        partial = sum(1 for r in results if self.device_result_status(r) == "PARTIAL")
        failed = len(results) - connected
        summary_log, summary_json = self.save_batch_summary(results, context)
        self.qlog(f"\n{'-'*78}")
        self.qlog(f"Batch complete: {successful} successful | {partial} partial | {failed} connection failed")
        self.qlog(f"Summary LOG : {summary_log}")
        self.qlog(f"Manifest JSON: {summary_json}")
        self.qlog(f"{'-'*78}")

        artifact_paths = [summary_log, summary_json]
        for result in results:
            saved = result.get("_saved_files", {})
            artifact_paths.extend([saved.get("log"), saved.get("json"), saved.get("running_config")])
        collection_anchor = common_artifact_directory(artifact_paths, context["storage_base"])

        state = "success" if successful == len(results) else ("error" if connected == 0 else "warning")
        return {
            "state": state,
            "status": f"Collection Complete — {successful} successful, {partial} partial, {failed} failed",
            "last_collection_dir": str(collection_anchor),
        }

    def collect_device(self, device, commands, profile_metadata, context):
        started_dt = datetime.now().astimezone()
        started_perf = time.perf_counter()
        conn = None
        connection = {
            "status": "connection_error",
            "success": False,
            "error_type": "",
            "error": "",
            "credential_source": self._run_credential_source,
            "username": self._run_credentials["username"] if self._run_credentials else "",
            "token_used": bool(self._run_credentials and self._run_credentials.get("token_used")),
            "netmiko_device_type": device.get("device_type", self.settings.get("default_device_type", "cisco_ios")) or "cisco_ios",
            "detected_prompt": "",
            "tcp_22_precheck": {
                "enabled": bool(self.settings.get("tcp_22_precheck", True)),
                "status": "not_run",
                "duration_seconds": None,
                "error": "",
            },
        }
        result = {
            "schema_version": SCHEMA_VERSION,
            "schema_name": SCHEMA_NAME,
            "document_type": "device_collection",
            "collector": {
                "name": APP_NAME,
                "version": COLLECTOR_VERSION,
            },
            "collection": {
                "mode": "full",
                "collection_id": context["collection_id"],
                "batch_started": context["started"],
                "started": started_dt.isoformat(timespec="milliseconds"),
                "finished": None,
                "duration_seconds": None,
                "profile": context["profile_info"],
            },
            "device": {
                "name": device.get("name", ""),
                "ip": device.get("ip", ""),
                "location": device.get("location", device.get("group", "")),
                "role": device.get("role", ""),
                "notes": device.get("notes", ""),
                "configured_device_type": device.get("device_type", "cisco_ios"),
            },
            "connection": connection,
            "commands": [],
            "artifacts": {
                "running_config": {
                    "available": False,
                    "filename": None,
                    "source_command_sequence": None,
                    "source_command": None,
                    "sha256": None,
                    "bytes": 0,
                    "format": "text/plain; cisco-running-config",
                }
            },
            "summary": {
                "overall_status": "connection_failed",
                "commands_planned": len(commands),
                "commands_attempted": 0,
                "commands_successful": 0,
                "commands_failed": 0,
            },
        }

        name = device.get("name") or device.get("ip")
        ip = device.get("ip")
        self.qlog(f"\n>>> {name} ({ip})")

        try:
            if connection["tcp_22_precheck"]["enabled"]:
                open22, elapsed, reason = tcp_22_test(ip)
                connection["tcp_22_precheck"]["duration_seconds"] = round(elapsed, 3)
                if not open22:
                    connection["tcp_22_precheck"]["status"] = "failed"
                    connection["tcp_22_precheck"]["error"] = reason
                    connection["status"] = "tcp_22_unreachable"
                    connection["error_type"] = "ConnectionError"
                    connection["error"] = f"TCP/22 unreachable after {elapsed:.2f}s: {reason}"
                    raise ConnectionError(connection["error"])
                connection["tcp_22_precheck"]["status"] = "success"
                self.qlog(f"    [PASS] TCP/22 reachable ({elapsed:.2f}s)")

            conn, info = self.connect_device(device)
            connection["status"] = "success"
            connection["success"] = True
            connection["detected_prompt"] = info["prompt"]
            connection["netmiko_device_type"] = info["device_type"]
            self.qlog(f"    [PASS] SSH connected | prompt={info['prompt'] or '(unknown)'} | auth={info['credential_source']}")

            # Terminal setup is intentionally outside the requested command set and is not
            # written into command output. Failure here never aborts a collection.
            try:
                conn.send_command_timing("terminal length 0", strip_prompt=True, strip_command=True)
            except Exception:
                pass
            try:
                conn.send_command_timing("terminal width 511", strip_prompt=True, strip_command=True)
            except Exception:
                pass

            read_timeout = profile_metadata.get("read_timeout", self.settings.get("command_timeout", 45))
            for number, cmd in enumerate(commands, 1):
                cmd_started_dt = datetime.now().astimezone()
                cmd_started_perf = time.perf_counter()
                entry = {
                    "sequence": number,
                    "command": cmd,
                    "status": "execution_error",
                    "success": False,
                    "started": cmd_started_dt.isoformat(timespec="milliseconds"),
                    "finished": None,
                    "duration_seconds": None,
                    "output": "",
                    "error": "",
                }
                self.qlog(f"    [{number:02d}/{len(commands):02d}] {cmd}")
                try:
                    # OUTPUT CONTRACT: Never parse, sort, normalize, or reinterpret this string.
                    # Prompt and command echo are stripped deliberately so commands[].output is
                    # the reusable response body for downstream scripts.
                    output = conn.send_command(
                        cmd,
                        strip_prompt=True,
                        strip_command=True,
                        read_timeout=float(read_timeout),
                    )
                    entry["output"] = output
                    if cli_output_indicates_error(output):
                        entry["status"] = "cli_error"
                        entry["error"] = "Device returned a CLI command error."
                        self.qlog("         [FAIL] CLI command error; continuing.")
                    else:
                        entry["status"] = "success"
                        entry["success"] = True
                        self.qlog("         [PASS]")
                except Exception as exc:
                    entry["status"] = classify_command_exception(exc)
                    entry["error"] = self.clean_exception(exc)
                    self.qlog(f"         [FAIL] {entry['error']} | continuing.")
                finally:
                    entry["finished"] = local_timestamp()
                    entry["duration_seconds"] = round(time.perf_counter() - cmd_started_perf, 3)
                    result["commands"].append(entry)

        except NetmikoAuthenticationException as exc:
            connection["status"] = "authentication_failed"
            connection["error_type"] = exc.__class__.__name__
            connection["error"] = f"Authentication failed: {self.clean_exception(exc)}"
            self.qlog(f"    [FAIL] {connection['error']}")
        except NetmikoTimeoutException as exc:
            connection["status"] = "timeout"
            connection["error_type"] = exc.__class__.__name__
            connection["error"] = f"SSH timed out: {self.clean_exception(exc)}"
            self.qlog(f"    [FAIL] {connection['error']}")
        except Exception as exc:
            # Preserve a more specific status already set by the TCP/22 precheck.
            if connection["status"] == "connection_error":
                connection["error_type"] = exc.__class__.__name__
                connection["error"] = self.clean_exception(exc)
            self.qlog(f"    [FAIL] {connection['error'] or self.clean_exception(exc)}")
        finally:
            if conn:
                try:
                    conn.disconnect()
                except Exception:
                    pass

            result["collection"]["finished"] = local_timestamp()
            result["collection"]["duration_seconds"] = round(time.perf_counter() - started_perf, 3)
            attempted = len(result["commands"])
            succeeded = sum(1 for x in result["commands"] if x.get("status") == "success")
            failed = attempted - succeeded
            if not connection["success"]:
                overall = "connection_failed"
            elif failed:
                overall = "partial"
            else:
                overall = "success"
            result["summary"] = {
                "overall_status": overall,
                "commands_planned": len(commands),
                "commands_attempted": attempted,
                "commands_successful": succeeded,
                "commands_failed": failed,
            }

            try:
                log_path, json_path, cfg_path = self.save_result(result, context)
                result["_saved_files"] = {"log": log_path, "json": json_path, "running_config": cfg_path}
                self.qlog(f"    Saved LOG : {log_path}")
                self.qlog(f"    Saved JSON: {json_path}")
                if cfg_path:
                    self.qlog(f"    Saved CFG : {cfg_path}")
            except Exception as exc:
                self.qlog(f"    [FAIL] Could not save output: {self.clean_exception(exc)}")
        return result

    def save_result(self, result, context):
        device_name = safe_filename(result["device"].get("name") or result["device"].get("ip"))
        device_values = storage_token_values(
            context.get("run_datetime", datetime.now().astimezone()),
            context.get("profile", ""),
            context.get("collection_id", ""),
            result.get("device", {}),
        )
        storage_layout = normalize_storage_layout(context.get("storage_layout", DEFAULT_STORAGE_LAYOUT))
        levels = storage_layout.get("levels", [])
        device_depth = storage_layout.get("destinations", {}).get("device_files", len(levels))
        device_dir = build_storage_path_to_depth(
            context["storage_base"], levels, device_depth, device_values
        )
        device_dir.mkdir(parents=True, exist_ok=True)

        base = f"{device_name}_{context['safe_profile']}_{context['collection_id']}"
        log_path = device_dir / f"{base}.log"
        json_path = device_dir / f"{base}.json"

        cfg_path = None
        cfg_entry = find_running_config_entry(result)
        if cfg_entry is not None:
            cfg_text = build_running_config_cfg(cfg_entry.get("output", ""))
            if cfg_text.strip():
                cfg_path = device_dir / f"{device_name}_running-config_{context['collection_id']}.cfg"
                cfg_path.write_text(cfg_text, encoding="utf-8")
                cfg_bytes = cfg_text.encode("utf-8")
                result["artifacts"]["running_config"] = {
                    "available": True,
                    "filename": cfg_path.name,
                    "source_command_sequence": cfg_entry.get("sequence"),
                    "source_command": cfg_entry.get("command"),
                    "sha256": sha256_bytes(cfg_bytes),
                    "bytes": len(cfg_bytes),
                    "format": "text/plain; cisco-running-config",
                }

        # Device JSON remains the canonical machine-readable record. LOG and CFG are
        # sibling human/recovery artifacts by design.
        atomic_write_json(json_path, result)
        log_path.write_text(self.render_universal_log(result), encoding="utf-8")
        return log_path, json_path, cfg_path

    @staticmethod
    def device_result_status(result):
        status = result.get("summary", {}).get("overall_status", "connection_failed")
        return {
            "success": "SUCCESS",
            "partial": "PARTIAL",
            "connection_failed": "SSH FAILED",
        }.get(status, "PARTIAL")

    def save_batch_summary(self, results, context):
        storage_layout = normalize_storage_layout(context.get("storage_layout", DEFAULT_STORAGE_LAYOUT))
        levels = storage_layout.get("levels", [])
        destinations = storage_layout.get("destinations", {})
        collection_values = storage_token_values(
            context.get("run_datetime", datetime.now().astimezone()),
            context.get("profile", ""),
            context.get("collection_id", ""),
            {},
        )

        manifest_dir = build_storage_path_to_depth(
            context["storage_base"], levels, destinations.get("manifest", 0), collection_values
        )
        summary_dir = build_storage_path_to_depth(
            context["storage_base"], levels, destinations.get("summary", 0), collection_values
        )
        manifest_dir.mkdir(parents=True, exist_ok=True)
        summary_dir.mkdir(parents=True, exist_ok=True)

        summary_base = f"Collection_Summary_{context['safe_profile']}_{context['collection_id']}"
        manifest_base = f"Collection_Manifest_{context['safe_profile']}_{context['collection_id']}"
        json_path = manifest_dir / f"{manifest_base}.json"
        log_path = summary_dir / f"{summary_base}.log"

        def manifest_relative(path):
            if not path:
                return ""
            try:
                return os.path.relpath(str(Path(path)), start=str(manifest_dir))
            except Exception:
                return str(path)

        rows = []
        for result in results:
            saved = result.get("_saved_files", {})
            summary = result.get("summary", {})
            connection = result.get("connection", {})
            rows.append({
                "name": result.get("device", {}).get("name", ""),
                "ip": result.get("device", {}).get("ip", ""),
                "location": result.get("device", {}).get("location", ""),
                "status": summary.get("overall_status", "connection_failed"),
                "connection_status": connection.get("status", "connection_error"),
                "commands_planned": summary.get("commands_planned", 0),
                "commands_attempted": summary.get("commands_attempted", 0),
                "commands_successful": summary.get("commands_successful", 0),
                "commands_failed": summary.get("commands_failed", 0),
                "connection_error": connection.get("error", ""),
                "files": {
                    "log": manifest_relative(saved.get("log")),
                    "json": manifest_relative(saved.get("json")),
                    "running_config": manifest_relative(saved.get("running_config")) or None,
                },
            })

        successful = sum(1 for row in rows if row["status"] == "success")
        partial = sum(1 for row in rows if row["status"] == "partial")
        failed = sum(1 for row in rows if row["status"] == "connection_failed")
        finished = datetime.now().astimezone()
        try:
            started_dt = datetime.fromisoformat(context["started"])
            batch_duration = round((finished - started_dt).total_seconds(), 3)
        except Exception:
            batch_duration = None

        manifest = {
            "schema_version": SCHEMA_VERSION,
            "schema_name": SCHEMA_NAME,
            "document_type": "collection_manifest",
            "collector": {
                "name": APP_NAME,
                "version": COLLECTOR_VERSION,
            },
            "collection": {
                "collection_id": context["collection_id"],
                "started": context["started"],
                "finished": finished.isoformat(timespec="milliseconds"),
                "duration_seconds": batch_duration,
                "profile": context["profile_info"],
                "storage": {
                    "layout_name": context.get("storage_layout_name", "Custom"),
                    "relative_path_base": "manifest_directory",
                },
            },
            "artifacts": {
                "summary_log": manifest_relative(log_path),
            },
            "summary": {
                "devices_selected": len(rows),
                "connected": len(rows) - failed,
                "successful": successful,
                "partial": partial,
                "connection_failed": failed,
            },
            "devices": rows,
        }

        atomic_write_json(json_path, manifest)
        log_path.write_text(self.render_batch_summary(manifest), encoding="utf-8")
        return log_path, json_path

            f"Commands Failed     : {summary.get('commands_failed', 0)}",
            "=" * 92,
            "",
        ])
        return "\n".join(lines)

    # -------------------- Worker / UI --------------------

    def start_worker(self, func, *args, operation_label="Working..."):
        if self.busy:
            messagebox.showinfo("Collector busy", "A collection or test is already running.")
            return
        self.busy = True
        self.progress["value"] = 0
        self.set_operation_visual("neutral", operation_label)
        self.status_var.set(operation_label)
        self.update_action_states()
        threading.Thread(target=self._worker_wrapper, args=(func, args), daemon=True).start()

    def _worker_wrapper(self, func, args):
        result = None
        try:
            result = func(*args)
        except Exception as exc:
            self.ui_queue.put(("log", f"[UNHANDLED ERROR] {self.clean_exception(exc)}"))
            result = {"state": "error", "status": "Operation Failed"}
        finally:
            self.ui_queue.put(("done", result or {"state": "success", "status": "Complete"}))

    def qlog(self, text):
        self.ui_queue.put(("log", text))

    def qprogress(self, value):
        self.ui_queue.put(("progress", value))

    def process_ui_queue(self):
        try:
            while True:
                kind, value = self.ui_queue.get_nowait()
                if kind == "log":
                    self.log_ui(value)
                elif kind == "progress":
                    self.progress["value"] = value
                elif kind == "done":
                    self.busy = False
                    self.progress["value"] = 100
                    result = value or {}
                    state = result.get("state", "success")
                    status = result.get("status", "Complete")
                    self.set_operation_visual(state, status)
                    self.status_var.set(status)
                    last_dir = result.get("last_collection_dir")
                    if last_dir:
                        self.last_collection_dir = Path(last_dir)
                        self.settings["last_collection_directory"] = str(self.last_collection_dir)
                        atomic_write_json(SETTINGS_FILE, self.settings)
                        self._refresh_last_collection_menu()
                    self.clear_run_credentials()
                    self.update_action_states()
        except queue.Empty:
            pass
        self.after(100, self.process_ui_queue)

    def log_ui(self, text):
        stamp = datetime.now().strftime("%H:%M:%S")
        self.activity.insert("end", f"[{stamp}] {text}\n")
        self.activity.see("end")

    @staticmethod
    def clean_exception(exc):
        text = str(exc).strip().replace("\r", " ").replace("\n", " ")
        return re.sub(r"\s+", " ", text)[:700]

    def open_path(self, path: Path):
        """Open a file or folder with resilient OS-specific fallbacks."""
        path = Path(path).expanduser()

        try:
            if not path.exists() and not path.suffix:
                path.mkdir(parents=True, exist_ok=True)
        except Exception:
            pass

        target = os.path.normpath(str(path))
        system = platform.system().lower()
        attempts = []

        def attempt(label, launcher):
            try:
                launcher()
                self.log_ui(f"[OPEN] {label}: {target}")
                return True
            except Exception as exc:
                attempts.append(f"{label}: {self.clean_exception(exc)}")
                return False

        if system == "windows":
            flags = getattr(subprocess, "CREATE_NO_WINDOW", 0)

            if path.is_dir():
                # Direct Explorer launch is reliable on managed Windows devices and
                # supports local paths as well as UNC/network paths.
                if attempt(
                    "Windows Explorer",
                    lambda: subprocess.Popen(
                        ["explorer.exe", target],
                        creationflags=flags,
                    ),
                ):
                    return

                if hasattr(os, "startfile") and attempt(
                    "Windows shell",
                    lambda: os.startfile(target),
                ):
                    return

                if attempt(
                    "Windows command shell",
                    lambda: subprocess.Popen(
                        ["cmd.exe", "/c", "start", "", target],
                        creationflags=flags,
                    ),
                ):
                    return
            else:
                # Open files using the workstation's registered application first.
                if hasattr(os, "startfile") and attempt(
                    "Windows file association",
                    lambda: os.startfile(target),
                ):
                    return

                if attempt(
                    "Windows FileProtocolHandler",
                    lambda: subprocess.Popen(
                        ["rundll32.exe", "url.dll,FileProtocolHandler", target],
                        creationflags=flags,
                    ),
                ):
                    return

                # Collector artifacts and command profiles are plain text, so
                # Notepad is a dependable final direct-open fallback.
                if path.suffix.lower() in {".txt", ".log", ".json", ".cfg"} and attempt(
                    "Notepad fallback",
                    lambda: subprocess.Popen(
                        ["notepad.exe", target],
                        creationflags=flags,
                    ),
                ):
                    return

                # Even if file association launch is blocked, reveal the file.
                if attempt(
                    "Windows Explorer select",
                    lambda: subprocess.Popen(
                        ["explorer.exe", f"/select,{target}"],
                        creationflags=flags,
                    ),
                ):
                    return

        elif system == "darwin":
            if attempt("macOS open", lambda: subprocess.Popen(["open", target])):
                return
        else:
            if attempt("xdg-open", lambda: subprocess.Popen(["xdg-open", target])):
                return

        # If workstation policy blocks every launch mechanism, copy the path so
        # the menu/button still gives the engineer an immediately usable result.
        try:
            self.clipboard_clear()
            self.clipboard_append(target)
            self.update_idletasks()
            clipboard_note = "\n\nThe path has been copied to your clipboard."
        except Exception:
            clipboard_note = ""

        detail = "\n".join(attempts) if attempts else "No supported launch method succeeded."
        messagebox.showerror(
            "Could not open path",
            f"The operating system would not open:\n\n{target}"
            f"{clipboard_note}\n\nAttempts:\n{detail}",
        )
