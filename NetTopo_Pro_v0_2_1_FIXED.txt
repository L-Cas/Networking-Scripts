# ---------------------------- Data model ----------------------------


def new_id(prefix: str) -> str:
    return f"{prefix}_{uuid.uuid4().hex[:12]}"


@dataclass
class Device:
    id: str = field(default_factory=lambda: new_id("dev"))
    hostname: str = "NEW-DEVICE"
    serial: str = ""
    loopback: str = ""
    program: str = ""
    site: str = ""
    room: str = ""
    type: str = "Switch"
    model: str = ""
    active: bool = True
    notes: str = ""
    x: float = 0.0
    y: float = 0.0
    topology_role: str = "Auto"
    layout_priority: int = 0
    preferred_export_root: bool = False


@dataclass
class Network:
    id: str = field(default_factory=lambda: new_id("net"))
    name: str = "Network"
    vlan: str = ""
    notes: str = ""


@dataclass
class Link:
    id: str = field(default_factory=lambda: new_id("link"))
    a: str = ""  # internal device id
    a_interface: str = ""
    b: str = ""  # internal device id
    b_interface: str = ""
    media: str = "Fiber"
    speed: str = ""
    mode: str = "Trunk"
    active: bool = True
    notes: str = ""
    network_ids: list[str] = field(default_factory=list)
    next_hop_a_to_b: str = ""
    next_hop_b_to_a: str = ""
    show_annotation: bool = False


@dataclass
class ProjectData:
    schema_version: int = SCHEMA_VERSION
    devices: list[Device] = field(default_factory=list)
    links: list[Link] = field(default_factory=list)
    networks: list[Network] = field(default_factory=list)

    def device_by_id(self, device_id: str) -> Optional[Device]:
        return next((d for d in self.devices if d.id == device_id), None)

    def network_by_id(self, network_id: str) -> Optional[Network]:
        return next((n for n in self.networks if n.id == network_id), None)

    def link_by_id(self, link_id: str) -> Optional[Link]:
        return next((l for l in self.links if l.id == link_id), None)

    def hostname(self, device_id: str) -> str:
        dev = self.device_by_id(device_id)
        return dev.hostname if dev else "<missing>"

    def to_dict(self) -> dict:
        # Keep the human-readable v0.1-style endpoint fields while also persisting IDs.
        devices = [asdict(d) for d in self.devices]
        networks = [asdict(n) for n in self.networks]
        links = []
        memberships = []
        for link in self.links:
            row = asdict(link)
            row["a_id"] = link.a
            row["b_id"] = link.b
            row["a"] = self.hostname(link.a)
            row["b"] = self.hostname(link.b)
            row.pop("network_ids", None)
            links.append(row)
            for net_id in link.network_ids:
                net = self.network_by_id(net_id)
                memberships.append({
                    "link": link.id,
                    "network": net_id,
                    "link_name": f"{self.hostname(link.a)} ↔ {self.hostname(link.b)}",
                    "network_name": net.name if net else "",
                })
        return {
            "app": APP_NAME,
            "app_version": APP_VERSION,
            "schema_version": self.schema_version,
            "devices": devices,
            "links": links,
            "networks": networks,
            "memberships": memberships,
        }

    @classmethod
    def from_dict(cls, payload: dict) -> "ProjectData":
        project = cls(schema_version=int(payload.get("schema_version", payload.get("version", 1) or 1)))

        # Devices: accept old v0.1 data without IDs / newer fields.
        for raw in payload.get("devices", []):
            if not isinstance(raw, dict):
                continue
            allowed = {f.name for f in Device.__dataclass_fields__.values()}
            kwargs = {k: v for k, v in raw.items() if k in allowed}
            if not kwargs.get("id"):
                kwargs["id"] = new_id("dev")
            project.devices.append(Device(**kwargs))

        # Networks: accept id-less rows.
        for raw in payload.get("networks", []):
            if not isinstance(raw, dict):
                continue
            allowed = {f.name for f in Network.__dataclass_fields__.values()}
            kwargs = {k: v for k, v in raw.items() if k in allowed}
            if not kwargs.get("id"):
                kwargs["id"] = new_id("net")
            project.networks.append(Network(**kwargs))

        host_to_id = {d.hostname.lower(): d.id for d in project.devices}
        network_name_to_id = {n.name.lower(): n.id for n in project.networks}

        # Links: resolve endpoints from IDs first, then v0.1 hostname fields.
        for raw in payload.get("links", []):
            if not isinstance(raw, dict):
                continue
            a_value = str(raw.get("a_id") or raw.get("a") or "")
            b_value = str(raw.get("b_id") or raw.get("b") or "")
            a_id = a_value if project.device_by_id(a_value) else host_to_id.get(a_value.lower(), "")
            b_id = b_value if project.device_by_id(b_value) else host_to_id.get(b_value.lower(), "")
            if not a_id or not b_id:
                continue
            link = Link(
                id=str(raw.get("id") or new_id("link")),
                a=a_id,
                a_interface=str(raw.get("a_interface", "")),
                b=b_id,
                b_interface=str(raw.get("b_interface", "")),
                media=str(raw.get("media", "Fiber")),
                speed=str(raw.get("speed", "")),
                mode=str(raw.get("mode", "Trunk")),
                active=bool(raw.get("active", True)),
                notes=str(raw.get("notes", "")),
                next_hop_a_to_b=str(raw.get("next_hop_a_to_b", "")),
                next_hop_b_to_a=str(raw.get("next_hop_b_to_a", "")),
                show_annotation=bool(raw.get("show_annotation", False)),
            )
            # Some transitional builds stored network_ids directly.
            direct_ids = raw.get("network_ids", [])
            if isinstance(direct_ids, list):
                link.network_ids.extend([str(v) for v in direct_ids if project.network_by_id(str(v))])
            project.links.append(link)

        link_by_id = {l.id: l for l in project.links}
        # Memberships from v0.1/v0.2 human-readable model.
        for raw in payload.get("memberships", []):
            if not isinstance(raw, dict):
                continue
            link_id = str(raw.get("link") or raw.get("link_id") or "")
            net_value = str(raw.get("network") or raw.get("network_id") or raw.get("network_name") or "")
            link = link_by_id.get(link_id)
            net_id = net_value if project.network_by_id(net_value) else network_name_to_id.get(net_value.lower(), "")
            if link and net_id and net_id not in link.network_ids:
                link.network_ids.append(net_id)

        return project

class DeviceItem(QGraphicsRectItem):
    WIDTH = 156
    HEIGHT = 70

    def __init__(self, controller: "MainWindow", device_id: str):
        super().__init__(-self.WIDTH / 2, -self.HEIGHT / 2, self.WIDTH, self.HEIGHT)
        self.controller = controller
        self.device_id = device_id
        self.setFlags(
            QGraphicsItem.GraphicsItemFlag.ItemIsMovable
            | QGraphicsItem.GraphicsItemFlag.ItemIsSelectable
            | QGraphicsItem.GraphicsItemFlag.ItemSendsGeometryChanges
        )
        self.setZValue(10)
        self.setPen(QPen(QColor("#c5cbd2"), 1.2))
        self.label = QGraphicsSimpleTextItem(self)
        self.label.setBrush(QBrush(QColor("#f5f7fa")))
        font = QFont()
        font.setPointSize(9)
        font.setBold(True)
        self.label.setFont(font)
        self.refresh()

    def refresh(self):
        device = self.controller.project.device_by_id(self.device_id)
        if not device:
            return
        self.setBrush(QBrush(ROLE_COLORS.get(device.topology_role, ROLE_COLORS["Auto"])))
        role_text = device.topology_role if device.topology_role != "Auto" else "Auto role"
        model_text = device.model or device.type
        text = f"{device.hostname}\n{model_text}  •  {role_text}"
        self.label.setText(text)
        rect = self.label.boundingRect()
        self.label.setPos(-rect.width() / 2, -rect.height() / 2)
        self.setOpacity(1.0 if device.active else 0.45)

    def mousePressEvent(self, event):
        self.controller.canvas_device_clicked(self.device_id)
        super().mousePressEvent(event)

    def mouseDoubleClickEvent(self, event):
        self.controller.show_device_properties(self.device_id)
        super().mouseDoubleClickEvent(event)

    def itemChange(self, change, value):
        if change == QGraphicsItem.GraphicsItemChange.ItemPositionHasChanged:
            device = self.controller.project.device_by_id(self.device_id)
            if device:
                pos = self.pos()
                device.x = float(pos.x())
                device.y = float(pos.y())
                self.controller.update_links_for_device(self.device_id)
                self.controller.set_dirty(True)
        if change == QGraphicsItem.GraphicsItemChange.ItemSelectedHasChanged and bool(value):
            self.controller.show_device_properties(self.device_id)
        return super().itemChange(change, value)


class LinkItem(QGraphicsPathItem):
    def __init__(self, controller: "MainWindow", link_id: str):
        super().__init__()
        self.controller = controller
        self.link_id = link_id
        self.setFlags(QGraphicsItem.GraphicsItemFlag.ItemIsSelectable)
        self.setZValue(2)
        self.label = QGraphicsSimpleTextItem(self)
        self.label.setBrush(QBrush(QColor("#d6d9dd")))
        self.label.setZValue(3)
        self.refresh()

    def refresh(self):
        link = self.controller.project.link_by_id(self.link_id)
        if not link:
            return
        a_item = self.controller.device_items.get(link.a)
        b_item = self.controller.device_items.get(link.b)
        if not a_item or not b_item:
            self.setPath(QPainterPath())
            return
        a = a_item.scenePos()
        b = b_item.scenePos()
        path = QPainterPath(a)
        path.lineTo(b)
        self.setPath(path)
        pen = QPen(QColor("#8eb7df") if link.active else QColor("#777b80"), 2.0)
        if not link.active:
            pen.setStyle(Qt.PenStyle.DashLine)
        self.setPen(pen)
        text_bits = []
        if link.a_interface or link.b_interface:
            text_bits.append(f"{link.a_interface or '?'} ↔ {link.b_interface or '?'}")
        if link.speed:
            text_bits.append(link.speed)
        self.label.setText("  |  ".join(text_bits))
        center = (a + b) / 2
        rect = self.label.boundingRect()
        self.label.setPos(center.x() - rect.width() / 2, center.y() - rect.height() - 5)

    def mousePressEvent(self, event):
        self.controller.show_link_properties(self.link_id)
        super().mousePressEvent(event)

    def itemChange(self, change, value):
        if change == QGraphicsItem.GraphicsItemChange.ItemSelectedHasChanged and bool(value):
            self.controller.show_link_properties(self.link_id)
        return super().itemChange(change, value)


class TopologyView(QGraphicsView):
    def __init__(self, scene: QGraphicsScene, controller: "MainWindow"):
        super().__init__(scene)
        self.controller = controller
        self._space_down = False
        self._panning = False
        self._pan_start = None
        self.setRenderHints(QPainter.RenderHint.Antialiasing | QPainter.RenderHint.TextAntialiasing)
        self.setViewportUpdateMode(QGraphicsView.ViewportUpdateMode.BoundingRectViewportUpdate)
        self.setDragMode(QGraphicsView.DragMode.RubberBandDrag)
        self.setTransformationAnchor(QGraphicsView.ViewportAnchor.AnchorUnderMouse)
        self.setResizeAnchor(QGraphicsView.ViewportAnchor.AnchorViewCenter)
        self.setBackgroundBrush(QBrush(QColor("#1e2024")))
        self.setFocusPolicy(Qt.FocusPolicy.StrongFocus)

    def wheelEvent(self, event):
        factor = 1.18 if event.angleDelta().y() > 0 else 1 / 1.18
        current = self.transform().m11()
        new = current * factor
        if 0.2 <= new <= 4.0:
            self.scale(factor, factor)

    def keyPressEvent(self, event):
        if event.key() == Qt.Key.Key_Space and not event.isAutoRepeat():
            self._space_down = True
            self.setCursor(Qt.CursorShape.OpenHandCursor)
            event.accept()
            return
        super().keyPressEvent(event)

    def keyReleaseEvent(self, event):
        if event.key() == Qt.Key.Key_Space and not event.isAutoRepeat():
            self._space_down = False
            if not self._panning:
                self.unsetCursor()
            event.accept()
            return
        super().keyReleaseEvent(event)

    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.MiddleButton or (
            event.button() == Qt.MouseButton.LeftButton and self._space_down
        ):
            self._panning = True
            self._pan_start = event.position()
            self.setCursor(Qt.CursorShape.ClosedHandCursor)
            event.accept()
            return
        super().mousePressEvent(event)

    def mouseMoveEvent(self, event):
        if self._panning and self._pan_start is not None:
            delta = event.position() - self._pan_start
            self._pan_start = event.position()
            self.horizontalScrollBar().setValue(self.horizontalScrollBar().value() - int(delta.x()))
            self.verticalScrollBar().setValue(self.verticalScrollBar().value() - int(delta.y()))
            event.accept()
            return
        super().mouseMoveEvent(event)

    def mouseReleaseEvent(self, event):
        if self._panning and event.button() in (Qt.MouseButton.LeftButton, Qt.MouseButton.MiddleButton):
            self._panning = False
            self._pan_start = None
            self.setCursor(Qt.CursorShape.OpenHandCursor if self._space_down else Qt.CursorShape.ArrowCursor)
            event.accept()
            return
        super().mouseReleaseEvent(event)

class NetworkManagerDialog(QDialog):
    def __init__(self, project: ProjectData, on_changed: Callable[[], None], parent=None):
        super().__init__(parent)
        self.project = project
        self.on_changed = on_changed
        self.setWindowTitle("Networks")
        self.resize(680, 420)

        layout = QVBoxLayout(self)
        self.table = QTableWidget(0, 3)
        self.table.setHorizontalHeaderLabels(["Name", "VLAN", "Notes"])
        self.table.horizontalHeader().setStretchLastSection(True)
        layout.addWidget(self.table)

        row = QHBoxLayout()
        add_btn = QPushButton("Add Network")
        delete_btn = QPushButton("Delete Selected")
        row.addWidget(add_btn)
        row.addWidget(delete_btn)
        row.addStretch(1)
        layout.addLayout(row)

        buttons = QDialogButtonBox(QDialogButtonBox.StandardButton.Close)
        layout.addWidget(buttons)
        buttons.rejected.connect(self.reject)
        add_btn.clicked.connect(self.add_network)
        delete_btn.clicked.connect(self.delete_selected)
        self.table.itemChanged.connect(self.cell_changed)
        self.reload()

    def reload(self):
        self.table.blockSignals(True)
        self.table.setRowCount(len(self.project.networks))
        for r, net in enumerate(self.project.networks):
            for c, value in enumerate([net.name, net.vlan, net.notes]):
                item = QTableWidgetItem(value)
                item.setData(Qt.ItemDataRole.UserRole, net.id)
                self.table.setItem(r, c, item)
        self.table.blockSignals(False)

    def add_network(self):
        net = Network(name=f"Network {len(self.project.networks) + 1}")
        self.project.networks.append(net)
        self.reload()
        self.on_changed()

    def delete_selected(self):
        row = self.table.currentRow()
        if row < 0:
            return
        item = self.table.item(row, 0)
        if not item:
            return
        net_id = item.data(Qt.ItemDataRole.UserRole)
        self.project.networks = [n for n in self.project.networks if n.id != net_id]
        for link in self.project.links:
            link.network_ids = [nid for nid in link.network_ids if nid != net_id]
        self.reload()
        self.on_changed()

    def cell_changed(self, item: QTableWidgetItem):
        net_id = item.data(Qt.ItemDataRole.UserRole)
        net = self.project.network_by_id(net_id)
        if not net:
            return
        row = item.row()
        values = [self.table.item(row, c).text() if self.table.item(row, c) else "" for c in range(3)]
        net.name, net.vlan, net.notes = values
        self.on_changed()

    def showEvent(self, event):
        super().showEvent(event)
        self.fit()

    def _components(self, device_ids: list[str]) -> list[list[str]]:
        adj = {d: set() for d in device_ids}
        valid = set(device_ids)
        for link in self.project.links:
            if not link.active or link.a not in valid or link.b not in valid:
                continue
            adj[link.a].add(link.b)
            adj[link.b].add(link.a)
        components = []
        remaining = set(device_ids)
        while remaining:
            start = next(iter(remaining))
            stack = [start]
            comp = []
            remaining.remove(start)
            while stack:
                cur = stack.pop()
                comp.append(cur)
                for nb in adj[cur]:
                    if nb in remaining:
                        remaining.remove(nb)
                        stack.append(nb)
            components.append(comp)
        components.sort(key=lambda c: (-len(c), sorted(self.project.hostname(x).lower() for x in c)[0]))
        return components

    def _layout_component(self, comp: list[str], offset_x: float, offset_y: float) -> tuple[dict[str, QPointF], QRectF]:
        active_links = [l for l in self.project.links if l.active and l.a in comp and l.b in comp]
        adj = {d: set() for d in comp}
        for link in active_links:
            adj[link.a].add(link.b)
            adj[link.b].add(link.a)

        def role_rank(device_id: str) -> int:
            d = self.project.device_by_id(device_id)
            return ROLE_RANK.get(d.topology_role if d else "Auto", 5)

        roots = [d for d in comp if (self.project.device_by_id(d) and self.project.device_by_id(d).preferred_export_root)]
        if roots:
            root = sorted(roots, key=lambda d: (-self.project.device_by_id(d).layout_priority, self.project.hostname(d).lower()))[0]
        else:
            root = sorted(comp, key=lambda d: (role_rank(d), -len(adj[d]), -self.project.device_by_id(d).layout_priority, self.project.hostname(d).lower()))[0]

        levels = {root: 0}
        queue = [root]
        while queue:
            cur = queue.pop(0)
            for nb in sorted(adj[cur], key=lambda d: (role_rank(d), -self.project.device_by_id(d).layout_priority, self.project.hostname(d).lower())):
                if nb not in levels:
                    levels[nb] = levels[cur] + 1
                    queue.append(nb)
        for dev_id in comp:
            levels.setdefault(dev_id, max(levels.values(), default=0) + 1)

        # Respect explicit hierarchy roles as a minimum vertical level when they are set.
        for dev_id in comp:
            dev = self.project.device_by_id(dev_id)
            if dev and dev.topology_role not in ("Auto", "Unknown"):
                levels[dev_id] = max(levels[dev_id], ROLE_RANK.get(dev.topology_role, levels[dev_id]))

        grouped: dict[int, list[str]] = {}
        for dev_id, level in levels.items():
            grouped.setdefault(level, []).append(dev_id)

        h_gap = 235.0
        v_gap = 150.0
        positions: dict[str, QPointF] = {}
        max_width = 0.0
        max_level = 0
        for level in sorted(grouped):
            row = sorted(
                grouped[level],
                key=lambda d: (-self.project.device_by_id(d).layout_priority, self.project.hostname(d).lower()),
            )
            width = max(1, len(row) - 1) * h_gap
            max_width = max(max_width, width)
            start_x = offset_x - width / 2
            y = offset_y + level * v_gap
            for i, dev_id in enumerate(row):
                positions[dev_id] = QPointF(start_x + i * h_gap, y)
            max_level = max(max_level, level)

        bounds = QRectF(offset_x - max_width / 2 - 110, offset_y - 60, max_width + 220, max_level * v_gap + 140)
        return positions, bounds

    def build_scene(self):
        self.scene.clear()
        active_devices = [d for d in self.project.devices if d.active]
        if not active_devices:
            self.scene.addText("No active devices to export.")
            return

        positions: dict[str, QPointF] = {}
        x_cursor = 160.0
        top_y = 100.0
        component_gap = 180.0
        for comp in self._components([d.id for d in active_devices]):
            comp_positions, bounds = self._layout_component(comp, x_cursor + max(bounds_width_hint(comp), 400) / 2, top_y)
            # Normalize component to x_cursor.
            comp_rect = QRectF()
            for p in comp_positions.values():
                r = QRectF(p.x() - 90, p.y() - 40, 180, 80)
                comp_rect = r if comp_rect.isNull() else comp_rect.united(r)
            shift = QPointF(x_cursor - comp_rect.left(), 0)
            for dev_id, p in comp_positions.items():
                positions[dev_id] = p + shift
            x_cursor += comp_rect.width() + component_gap

        # Links first, devices second.
        for link in self.project.links:
            if not link.active or link.a not in positions or link.b not in positions:
                continue
            a = positions[link.a]
            b = positions[link.b]
            path = orthogonal_path(a, b)
            item = self.scene.addPath(path, QPen(QColor("#465463"), 2.0))
            item.setZValue(1)

            # Interface label near center.
            bits = []
            if link.a_interface or link.b_interface:
                bits.append(f"{link.a_interface or '?'} ↔ {link.b_interface or '?'}")
            if link.speed:
                bits.append(link.speed)
            if bits:
                label = self.scene.addSimpleText("  |  ".join(bits))
                label.setBrush(QBrush(QColor("#31373d")))
                font = QFont()
                font.setPointSize(8)
                label.setFont(font)
                center = path.pointAtPercent(0.5)
                br = label.boundingRect()
                label.setPos(center.x() - br.width() / 2, center.y() - br.height() - 3)
                label.setZValue(3)

            if link.show_annotation:
                annotations = []
                if link.next_hop_a_to_b:
                    annotations.append(f"A→B: {link.next_hop_a_to_b}")
                if link.next_hop_b_to_a:
                    annotations.append(f"B→A: {link.next_hop_b_to_a}")
                if annotations:
                    ann = self.scene.addSimpleText(" | ".join(annotations))
                    ann.setBrush(QBrush(QColor("#5b3f64")))
                    font = QFont()
                    font.setPointSize(7)
                    font.setItalic(True)
                    ann.setFont(font)
                    center = path.pointAtPercent(0.5)
                    br = ann.boundingRect()
                    ann.setPos(center.x() - br.width() / 2, center.y() + 5)
                    ann.setZValue(3)

        for dev in active_devices:
            p = positions[dev.id]
            rect = QRectF(p.x() - 90, p.y() - 38, 180, 76)
            role = dev.topology_role if dev.topology_role != "Auto" else "Auto"
            brush = QBrush(ROLE_COLORS.get(role, ROLE_COLORS["Auto"]).lighter(170))
            shape = self.scene.addRect(rect, QPen(QColor("#26313c"), 1.5), brush)
            shape.setZValue(5)
            label = self.scene.addSimpleText(
                f"{dev.hostname}\n{dev.type}{' • ' + dev.model if dev.model else ''}\n{role}"
            )
            label.setBrush(QBrush(QColor("#11161a")))
            font = QFont()
            font.setPointSize(8)
            font.setBold(True)
            label.setFont(font)
            br = label.boundingRect()
            label.setPos(p.x() - br.width() / 2, p.y() - br.height() / 2)
            label.setZValue(6)

    def _render_to_painter(self, painter: QPainter, target: QRectF):
        source = self.scene.itemsBoundingRect().adjusted(-35, -35, 35, 35)
        self.scene.render(painter, target, source, Qt.AspectRatioMode.KeepAspectRatio)

    def export_pdf(self):
        path, _ = QFileDialog.getSaveFileName(self, "Export PDF", "network_topology.pdf", "PDF (*.pdf)")
        if not path:
            return
        if not path.lower().endswith(".pdf"):
            path += ".pdf"
        printer = QPrinter(QPrinter.PrinterMode.HighResolution)
        printer.setOutputFormat(QPrinter.OutputFormat.PdfFormat)
        printer.setOutputFileName(path)
        printer.setPageOrientation(QPageLayout.Orientation.Landscape)
        painter = QPainter(printer)
        if not painter.isActive():
            QMessageBox.critical(self, "Export Failed", "Could not create the PDF painter.")

        target = QRectF(printer.pageRect(QPrinter.Unit.DevicePixel))
        self._render_to_painter(painter, target)
        painter.end()
        QMessageBox.information(self, "Export Complete", f"Saved PDF:\n{path}")

    def export_svg(self):
        path, _ = QFileDialog.getSaveFileName(self, "Export SVG", "network_topology.svg", "SVG (*.svg)")
        if not path:
            return

            path += ".svg"
        source = self.scene.itemsBoundingRect().adjusted(-35, -35, 35, 35)
        generator = QSvgGenerator()

        generator.setViewBox(source.toRect())

        generator.setDescription("Editable vector export generated by NetTopo Pro")

        self.scene.render(painter, QRectF(generator.viewBox()), source, Qt.AspectRatioMode.KeepAspectRatio)
        painter.end()
        QMessageBox.information(self, "Export Complete", f"Saved SVG:\n{path}")


def bounds_width_hint(comp: list[str]) -> float:
    # Used only to give the first pass a stable center before normalization.
    return max(400.0, min(1400.0, len(comp) * 170.0))


def orthogonal_path(a: QPointF, b: QPointF) -> QPainterPath:
    path = QPainterPath(a)
    dx = abs(a.x() - b.x())
    dy = abs(a.y() - b.y())
    if dy >= 35:
        mid_y = (a.y() + b.y()) / 2
        path.lineTo(a.x(), mid_y)
        path.lineTo(b.x(), mid_y)
        path.lineTo(b)
    elif dx >= 35:
        mid_x = (a.x() + b.x()) / 2
        path.lineTo(mid_x, a.y())
        path.lineTo(mid_x, b.y())
        path.lineTo(b)
    else:
        path.lineTo(b)
    return path

    # ---------- Project / file ----------

    def set_dirty(self, value: bool):
        self.dirty = value
        title = f"{APP_NAME} v{APP_VERSION}"
        if self.project_path:
            title += f" — {self.project_path.name}"
        if self.dirty:
            title += " *"
        self.setWindowTitle(title)

    def maybe_save(self) -> bool:
        if not self.dirty:
            return True
        answer = QMessageBox.question(
            self,
            "Unsaved Changes",
            "Save changes to the current topology?",
            QMessageBox.StandardButton.Save | QMessageBox.StandardButton.Discard | QMessageBox.StandardButton.Cancel,
            QMessageBox.StandardButton.Save,
        )
        if answer == QMessageBox.StandardButton.Cancel:
            return False
        if answer == QMessageBox.StandardButton.Save:
            return self.save_project()
        return True

    def new_project(self):
        if not self.maybe_save():
            return
        self.project = ProjectData()
        self.project_path = None
        self.connect_first_device = None
        self.refresh_all()
        self.set_dirty(False)
        self.statusBar().showMessage("New topology created.", 4000)

    def open_project(self):
        if not self.maybe_save():
            return
        path, _ = QFileDialog.getOpenFileName(
            self,
            "Open NetTopo Project",
            "",
            "NetTopo Project (*.ntopo *.json);;JSON (*.json);;All files (*.*)",
        )
        if not path:
            return
        try:
            payload = json.loads(Path(path).read_text(encoding="utf-8"))
            self.project = ProjectData.from_dict(payload)
            self.project_path = Path(path)
            self.refresh_all()
            self.set_dirty(False)
            self.statusBar().showMessage(f"Opened {path}", 5000)
        except Exception as exc:
            QMessageBox.critical(self, "Open Failed", f"Could not open the project:\n\n{exc}")

    def save_project(self, save_as: bool = False) -> bool:
        path = self.project_path
        if save_as or path is None:
            selected, _ = QFileDialog.getSaveFileName(
                self,
                "Save NetTopo Project",
                self.project_path.name if self.project_path else f"network_topology{PROJECT_EXT}",
                "NetTopo Project (*.ntopo);;JSON (*.json)",
            )
            if not selected:
                return False
            path = Path(selected)
            if not path.suffix:
                path = path.with_suffix(PROJECT_EXT)
        try:
            path.write_text(json.dumps(self.project.to_dict(), indent=2), encoding="utf-8")
            self.project_path = path
            self.set_dirty(False)
            self.statusBar().showMessage(f"Saved {path}", 5000)
            return True
        except Exception as exc:
            QMessageBox.critical(self, "Save Failed", f"Could not save the project:\n\n{exc}")
            return False

    # ---------- Refresh / canvas ----------

    def refresh_all(self):
        self.scene.clear()
        self.device_items.clear()
        self.link_items.clear()
        for device in self.project.devices:
            item = DeviceItem(self, device.id)
            item.setPos(device.x, device.y)
            self.scene.addItem(item)
            self.device_items[device.id] = item
        for link in self.project.links:
            if link.a in self.device_items and link.b in self.device_items:
                item = LinkItem(self, link.id)
                self.scene.addItem(item)
                self.link_items[link.id] = item
        self.refresh_device_list()
        self.clear_properties()

    def refresh_device_list(self):
        selected_ids = {i.data(Qt.ItemDataRole.UserRole) for i in self.device_list.selectedItems()}
        self.device_list.blockSignals(True)
        self.device_list.clear()
        for dev in sorted(self.project.devices, key=lambda d: d.hostname.lower()):
            item = QListWidgetItem(f"{dev.hostname}   [{dev.type}]")
            item.setData(Qt.ItemDataRole.UserRole, dev.id)
            self.device_list.addItem(item)
            if dev.id in selected_ids:
                item.setSelected(True)
        self.device_list.blockSignals(False)

    def add_device(self, dev_type: str = "Switch"):
        center = self.view.mapToScene(self.view.viewport().rect().center())
        base = dev_type.upper().replace(" ", "-")
        used = {d.hostname.lower() for d in self.project.devices}
        index = 1
        hostname = f"{base}-{index}"
        while hostname.lower() in used:
            index += 1
            hostname = f"{base}-{index}"
        dev = Device(hostname=hostname, type=dev_type, x=center.x(), y=center.y())
        self.project.devices.append(dev)
        item = DeviceItem(self, dev.id)
        item.setPos(dev.x, dev.y)
        self.scene.addItem(item)
        self.device_items[dev.id] = item
        self.refresh_device_list()
        self.show_device_properties(dev.id)
        item.setSelected(True)
        self.set_dirty(True)
        self.statusBar().showMessage(f"Added {dev_type}: {hostname}", 3500)

    def update_links_for_device(self, device_id: str):
        for link in self.project.links:
            if link.a == device_id or link.b == device_id:
                item = self.link_items.get(link.id)
                if item:
                    item.refresh()

        self.project.links.append(link)
        item = LinkItem(self, link.id)
        self.scene.addItem(item)
        self.link_items[link.id] = item
        self.show_link_properties(link.id)
        item.setSelected(True)
        self.set_dirty(True)
        self.statusBar().showMessage(f"Linked {self.project.hostname(a_id)} ↔ {self.project.hostname(device_id)}", 3500)
        # Stay in connect mode for rapid topology building.

    # ---------- Selection / deletion ----------

    def device_list_selection_changed(self):
        items = self.device_list.selectedItems()
        if not items:
            return
        device_id = items[0].data(Qt.ItemDataRole.UserRole)
        for item in self.scene.selectedItems():
            item.setSelected(False)
        graphics = self.device_items.get(device_id)
        if graphics:
            graphics.setSelected(True)
            self.view.centerOn(graphics)
        self.show_device_properties(device_id)

    def delete_selection(self):
        selected = self.scene.selectedItems()
        if not selected:
            list_items = self.device_list.selectedItems()
            if list_items:
                device_id = list_items[0].data(Qt.ItemDataRole.UserRole)
                item = self.device_items.get(device_id)
                selected = [item] if item else []
        if not selected:
            return

        devices_to_delete = {i.device_id for i in selected if isinstance(i, DeviceItem)}
        links_to_delete = {i.link_id for i in selected if isinstance(i, LinkItem)}
        if devices_to_delete:
            links_to_delete.update(
                l.id for l in self.project.links if l.a in devices_to_delete or l.b in devices_to_delete
            )
        count = len(devices_to_delete) + len(links_to_delete)
        answer = QMessageBox.question(
            self,
            "Delete Selection",
            f"Delete {count} selected topology object{'s' if count != 1 else ''}?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
            QMessageBox.StandardButton.No,
        )
        if answer != QMessageBox.StandardButton.Yes:
            return
        self.project.devices = [d for d in self.project.devices if d.id not in devices_to_delete]
        self.project.links = [l for l in self.project.links if l.id not in links_to_delete]
        self.refresh_all()
        self.set_dirty(True)

    # ---------- Properties ----------

    def clear_properties(self):
        while self.property_layout.count():
            item = self.property_layout.takeAt(0)
            widget = item.widget()
            if widget:
                widget.deleteLater()
        self.property_layout.addWidget(QLabel("Select a device or link to edit its properties."))
        self.property_layout.addStretch(1)

    def _reset_property_layout(self):
        while self.property_layout.count():
            item = self.property_layout.takeAt(0)
            widget = item.widget()
            if widget:
                widget.deleteLater()

    def _line(self, value: str, callback: Callable[[str], None], readonly: bool = False) -> QLineEdit:
        edit = QLineEdit(value)
        edit.setReadOnly(readonly)
        if not readonly:
            edit.editingFinished.connect(lambda e=edit: callback(e.text()))
        return edit

    def _combo(self, value: str, choices: list[str], callback: Callable[[str], None]) -> QComboBox:
        combo = QComboBox()
        combo.addItems(choices)
        if value not in choices:
            combo.addItem(value)
        combo.setCurrentText(value)
        combo.currentTextChanged.connect(callback)
        return combo

    def show_device_properties(self, device_id: str):
        device = self.project.device_by_id(device_id)
        if not device:
            return
        self._reset_property_layout()
        header = QLabel(f"Device — {device.hostname}")
        header.setStyleSheet("font-size:12pt;font-weight:700;")
        self.property_layout.addWidget(header)
        form = QFormLayout()

        def set_field(name: str, value):
            setattr(device, name, value)
            item = self.device_items.get(device.id)
            if item:
                item.refresh()
            self.refresh_device_list()
            self.set_dirty(True)

        form.addRow("Hostname", self._line(device.hostname, lambda v: set_field("hostname", v.strip() or device.hostname)))
        form.addRow("Serial", self._line(device.serial, lambda v: set_field("serial", v)))
        form.addRow("Loopback", self._line(device.loopback, lambda v: set_field("loopback", v)))
        form.addRow("Program", self._line(device.program, lambda v: set_field("program", v)))
        form.addRow("Site", self._line(device.site, lambda v: set_field("site", v)))
        form.addRow("Room / Closet", self._line(device.room, lambda v: set_field("room", v)))
        form.addRow("Device Type", self._combo(device.type, DEVICE_TYPES, lambda v: set_field("type", v)))
        form.addRow("Model", self._line(device.model, lambda v: set_field("model", v)))
        active = QCheckBox("Active")
        active.setChecked(device.active)
        active.toggled.connect(lambda v: set_field("active", bool(v)))
        form.addRow("Status", active)
        form.addRow("Topology Role", self._combo(device.topology_role, TOPOLOGY_ROLES, lambda v: set_field("topology_role", v)))
        priority = QSpinBox()
        priority.setRange(-1000, 1000)
        priority.setValue(device.layout_priority)
        priority.valueChanged.connect(lambda v: set_field("layout_priority", int(v)))
        form.addRow("Layout Priority", priority)
        root_box = QCheckBox("Preferred Export Root")
        root_box.setChecked(device.preferred_export_root)
        root_box.toggled.connect(lambda v: set_field("preferred_export_root", bool(v)))
        form.addRow("Export Root", root_box)
        form.addRow("Notes", self._line(device.notes, lambda v: set_field("notes", v)))
        self.property_layout.addLayout(form)
        self.property_layout.addStretch(1)

    def show_link_properties(self, link_id: str):
        link = self.project.link_by_id(link_id)
        if not link:
            return
        self._reset_property_layout()
        header = QLabel(f"Physical Link — {self.project.hostname(link.a)} ↔ {self.project.hostname(link.b)}")
        header.setWordWrap(True)
        header.setStyleSheet("font-size:12pt;font-weight:700;")
        self.property_layout.addWidget(header)
        form = QFormLayout()

        def set_field(name: str, value):
            setattr(link, name, value)
            item = self.link_items.get(link.id)
            if item:
                item.refresh()
            self.set_dirty(True)

        # Read-only endpoint fields use the intentionally darker v0.2.1 styling.
        form.addRow("Hostname A", self._line(self.project.hostname(link.a), lambda _v: None, readonly=True))
        form.addRow("Interface A", self._line(link.a_interface, lambda v: set_field("a_interface", v)))
        form.addRow("Hostname B", self._line(self.project.hostname(link.b), lambda _v: None, readonly=True))
        form.addRow("Interface B", self._line(link.b_interface, lambda v: set_field("b_interface", v)))
        form.addRow("Media", self._combo(link.media, MEDIA_TYPES, lambda v: set_field("media", v)))
        form.addRow("Speed", self._line(link.speed, lambda v: set_field("speed", v)))
        form.addRow("Mode", self._combo(link.mode, LINK_MODES, lambda v: set_field("mode", v)))
        active = QCheckBox("Active")
        active.setChecked(link.active)
        active.toggled.connect(lambda v: set_field("active", bool(v)))
        form.addRow("Status", active)
        form.addRow("Notes", self._line(link.notes, lambda v: set_field("notes", v)))
        form.addRow("Next Hop A→B", self._line(link.next_hop_a_to_b, lambda v: set_field("next_hop_a_to_b", v)))
        form.addRow("Next Hop B→A", self._line(link.next_hop_b_to_a, lambda v: set_field("next_hop_b_to_a", v)))
        show_ann = QCheckBox("Show directional annotation in export")
        show_ann.setChecked(link.show_annotation)
        show_ann.toggled.connect(lambda v: set_field("show_annotation", bool(v)))
        form.addRow("Annotation", show_ann)
        self.property_layout.addLayout(form)

        network_group = QGroupBox("Networks Carried")
        net_layout = QVBoxLayout(network_group)
        if not self.project.networks:
            empty = QLabel("No networks defined. Use Edit → Networks…")
            empty.setWordWrap(True)
            net_layout.addWidget(empty)
        else:
            for net in sorted(self.project.networks, key=lambda n: (safe_vlan_sort(n.vlan), n.name.lower())):
                box = QCheckBox(f"{net.vlan + ' — ' if net.vlan else ''}{net.name}")
                box.setChecked(net.id in link.network_ids)

                def toggle(checked: bool, net_id=net.id):
                    if checked and net_id not in link.network_ids:
                        link.network_ids.append(net_id)
                    if not checked and net_id in link.network_ids:
                        link.network_ids.remove(net_id)
                    self.set_dirty(True)

                box.toggled.connect(toggle)
                net_layout.addWidget(box)
        self.property_layout.addWidget(network_group)
        self.property_layout.addStretch(1)

def safe_vlan_sort(value: str):
    value = str(value).strip()
    try:
        return (0, int(value))
    except Exception:
        return (1, value.lower())
